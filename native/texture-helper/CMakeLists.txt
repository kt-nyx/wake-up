cmake_minimum_required(VERSION 3.21)
project(WakeUpTextureHelper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only x64 helpers are admitted")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(flag BUILD_TOOLS BUILD_SAMPLE BUILD_DX11 BUILD_DX12 BC_USE_OPENMP BUILD_SHARED_LIBS ENABLE_OPENEXR_SUPPORT ENABLE_LIBPNG_SUPPORT ENABLE_LIBJPEG_SUPPORT)
  set(${flag} OFF CACHE BOOL "S10 bounded CPU helper" FORCE)
endforeach()
if(NOT DIRECTXTEX_SOURCE OR NOT DIRECTXMATH_SOURCE)
  message(FATAL_ERROR "Use scripts/build_texture_helper.py for pinned source inputs")
endif()
add_subdirectory("${DIRECTXTEX_SOURCE}" directxtex)
# Explicitly prefer the reviewed Math headers over Windows SDK's bundled copy.
target_include_directories(DirectXTex BEFORE PRIVATE "${DIRECTXMATH_SOURCE}/Inc")
add_executable(wake-up-texture-helper main.cpp)
target_include_directories(wake-up-texture-helper BEFORE PRIVATE "${DIRECTXMATH_SOURCE}/Inc")
target_link_libraries(wake-up-texture-helper PRIVATE DirectXTex)
if(WIN32)
  target_link_libraries(wake-up-texture-helper PRIVATE windowscodecs ole32)
  target_compile_definitions(wake-up-texture-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(wake-up-texture-helper PRIVATE /W4 /guard:cf /Brepro)
  target_link_options(wake-up-texture-helper PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /Brepro)
else()
  find_package(PNG REQUIRED)
  find_package(JPEG REQUIRED)
  target_link_libraries(wake-up-texture-helper PRIVATE PNG::PNG JPEG::JPEG)
  target_compile_options(wake-up-texture-helper PRIVATE -Wall -Wextra -Wpedantic)
endif()
